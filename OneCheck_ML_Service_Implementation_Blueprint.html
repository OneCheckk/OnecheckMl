1. PDF Revision Stream Forensics (`services/pdf_forensics_service.py`)

Forensics
What It Does: Scans binary object offsets. If a college modified the date from 2021 to 2024 in Adobe Acrobat, it detects multiple /Root trailers and extracts the hidden original timestamp.
View Python Source Code 
import re

def analyze_pdf_revisions(pdf_bytes: bytes) -> dict:
    trailers = re.findall(rb'/Root', pdf_bytes)
    incremental_saves = len(trailers)
    
    modified = re.findall(rb'/ModDate\s*\((.*?)\)', pdf_bytes)
    created = re.findall(rb'/CreationDate\s*\((.*?)\)', pdf_bytes)
    creator = re.findall(rb'/Creator\s*\((.*?)\)', pdf_bytes)
    
    mod_str = modified[0].decode('latin1', errors='ignore') if modified else None
    create_str = created[0].decode('latin1', errors='ignore') if created else None
    tool_str = creator[0].decode('latin1', errors='ignore') if creator else "Standard Generator"
    
    is_tampered = incremental_saves > 1 and (mod_str != create_str)
    warning = None
    if is_tampered:
        warning = f"PDF modified after original issue! Tool detected: {tool_str}"
        
    return {
        "revision_count": incremental_saves,
        "is_tampered": is_tampered,
        "created_date": create_str,
        "modified_date": mod_str,
        "editor_tool": tool_str,
        "warning": warning
    }

2. Error Level Analysis (ELA) Heatmaps (`services/ela_service.py`)

Image Forensics
What It Does: Recompresses images at 90% JPEG quality. Pasted official stamps or edited signatures compress at a 10x higher error delta, generating glowing heatmaps.
View Python Source Code 
from PIL import Image
import numpy as np
import io, base64

def generate_ela_heatmap(image_bytes: bytes, quality: int = 90) -> dict:
    original = Image.open(io.BytesIO(image_bytes)).convert('RGB')
    buf = io.BytesIO()
    original.save(buf, 'JPEG', quality=quality)
    buf.seek(0)
    recompressed = Image.open(buf).convert('RGB')
    
    diff = np.abs(np.array(original, dtype=np.float32) - np.array(recompressed, dtype=np.float32))
    ela_amplified = (diff * 10).clip(0, 255).astype(np.uint8)
    
    gray_diff = np.mean(diff, axis=2)
    threshold = np.percentile(gray_diff, 95)
    suspicious_mask = gray_diff > threshold
    
    tampered_regions = []
    if suspicious_mask.any():
        from scipy import ndimage
        labeled, num_features = ndimage.label(suspicious_mask)
        for i in range(1, min(num_features + 1, 10)):
            r = np.where(labeled == i)
            if len(r[0]) > 50:
                tampered_regions.append({
                    "x": int(r[1].min()), "y": int(r[0].min()),
                    "width": int(r[1].max() - r[1].min()),
                    "height": int(r[0].max() - r[0].min()),
                    "confidence": round(float(gray_diff[labeled == i].mean()), 2)
                })
                
    suspicion_score = min(100, round(float(gray_diff.mean()) * 5, 1))
    
    h_buf = io.BytesIO()
    Image.fromarray(ela_amplified).save(h_buf, format='PNG')
    heatmap_b64 = base64.b64encode(h_buf.getvalue()).decode()
    
    return {
        "suspicion_score": suspicion_score,
        "tampered_regions": tampered_regions,
        "heatmap_base64": heatmap_b64
    }

4. 500m Geo-Fence & EXIF Spatial Verification (`services/geo_service.py`)

Spatial Forensics
What It Does: Extracts hidden camera GPS tags from photo headers and calculates geodesic distance from the registered campus. Distance > 500m triggers an instant FAIL flag.
View Python Source Code 
from PIL import Image
from PIL.ExifTags import TAGS, GPSTAGS
from geopy.distance import geodesic
import io

def extract_exif_gps_and_verify(image_bytes: bytes, campus_lat: float = None, campus_lng: float = None) -> dict:
    img = Image.open(io.BytesIO(image_bytes))
    exif = img._getexif()
    if not exif:
        return {"gps_lat": None, "gps_lng": None, "capture_date": None, "geofence_status": "NO_GPS_DATA", "distance_meters": None}
        
    gps_info = {}
    capture_date = None
    for tag_id, val in exif.items():
        name = TAGS.get(tag_id, tag_id)
        if name == "GPSInfo":
            for g_id, g_val in val.items():
                gps_info[GPSTAGS.get(g_id, g_id)] = g_val
        elif name == "DateTimeOriginal":
            capture_date = str(val)
            
    lat = lng = None
    if 'GPSLatitude' in gps_info and 'GPSLongitude' in gps_info:
        def dms_to_dec(dms, ref):
            deg = float(dms[0]) + float(dms[1])/60.0 + float(dms[2])/3600.0
            return -deg if ref in ['S', 'W'] else deg
        lat = round(dms_to_dec(gps_info['GPSLatitude'], gps_info.get('GPSLatitudeRef', 'N')), 6)
        lng = round(dms_to_dec(gps_info['GPSLongitude'], gps_info.get('GPSLongitudeRef', 'E')), 6)
        
    distance = None
    status = "NO_GPS_DATA"
    if lat and lng and campus_lat and campus_lng:
        distance = round(geodesic((campus_lat, campus_lng), (lat, lng)).meters, 1)
        status = "PASS" if distance <= 500.0 else "FAIL"
        
    return {
        "gps_lat": lat,
        "gps_lng": lng,
        "capture_date": capture_date,
        "distance_meters": distance,
        "geofence_status": status
    }

5. Tesseract OCR & Regulatory Regex Entity Extractor (`services/ocr_service.py`)

OCR
What It Does: Preprocesses images with Otsu thresholding and extracts certificate numbers, dates, and auto-classifies document types.
View Python Source Code 
import pytesseract
from PIL import Image
import io, re
from datetime import datetime

def extract_ocr_and_entities(image_bytes: bytes) -> dict:
    image = Image.open(io.BytesIO(image_bytes)).convert('L')
    data = pytesseract.image_to_data(image, output_type=pytesseract.Output.DICT)
    text = pytesseract.image_to_string(image)
    
    confidences = [int(c) for c in data['conf'] if int(c) > 0]
    avg_conf = sum(confidences) / len(confidences) if confidences else 0.0
    
    fields = {}
    dates = re.findall(r'\d{2}[-/]\d{2}[-/]\d{4}|\d{4}[-/]\d{2}[-/]\d{2}', text)
    if dates: fields['dates_found'] = dates
    
    cert_nums = re.findall(r'[A-Z]{2,}/[\w/]+/\d{4}/\d+', text)
    if cert_nums: fields['certificate_number'] = cert_nums[0]
    
    is_expired = False
    expiry_date = None
    for pattern in [r'valid.*?until[:\s]+(\d{2}[-/]\d{2}[-/]\d{4})', r'expir.*?[:\s]+(\d{2}[-/]\d{2}[-/]\d{4})']:
        m = re.search(pattern, text, re.IGNORECASE)
        if m:
            try:
                expiry_date = datetime.strptime(m.group(1), '%d-%m-%Y')
                is_expired = expiry_date < datetime.now()
                break
            except Exception: pass
            
    doc_type = 'OTHER'
    t_low = text.lower()
    if 'fire' in t_low and ('safety' in t_low or 'noc' in t_low): doc_type = 'FIRE_SAFETY_NOC'
    elif 'building completion' in t_low: doc_type = 'BUILDING_COMPLETION'
    elif 'affidavit' in t_low: doc_type = 'AFFIDAVIT'
    elif 'land' in t_low and 'record' in t_low: doc_type = 'LAND_RECORD'
    
    return {
        "text": text,
        "fields": fields,
        "confidence": round(avg_conf, 2),
        "classified_type": doc_type,
        "is_expired": is_expired,
        "expiry_date": expiry_date.isoformat() if expiry_date else None
    }

6. RapidFuzz Ghost Faculty Reconciliation (`services/ghost_service.py`)

Phonetic AI
What It Does: Token-sort phonetic matching detects ghost faculty under spelling variations (e.g., "Ramesh Sharma" vs "R. K. Sharma") across AICTE portal records without storing raw Aadhaar.
View Python Source Code 
from rapidfuzz import process, fuzz

def match_ghost_faculty(query_name: str, candidates: list, threshold: int = 90) -> list:
    cleaned_query = query_name.lower().strip()
    cleaned_candidates = [c.lower().strip() for c in candidates]
    
    results = process.extract(cleaned_query, cleaned_candidates, scorer=fuzz.token_sort_ratio)
    matches = []
    for candidate, score, idx in results:
        if score >= threshold:
            matches.append({
                "candidate": candidates[idx],
                "similarity_score": round(score, 1),
                "is_match": True
            })
    return matches

7. Validity-at-Decision-Date Engine (`services/validity_service.py`)

What It Does: Proactively alerts officers if a certificate (valid in March) will lapse before the academic term begins in August.
View Python Source Code 
from datetime import datetime

def evaluate_validity_at_date(expiry_date_str: str, decision_date_str: str = "2025-08-01") -> dict:
    expiry = datetime.fromisoformat(expiry_date_str.replace("Z", ""))
    decision = datetime.fromisoformat(decision_date_str.replace("Z", ""))
    
    lapsed_days = (decision - expiry).days
    if lapsed_days > 0:
        return {
            "status": "EXPIRED",
            "is_valid": False,
            "lapsed_days_before_term": lapsed_days,
            "alert": f"Certificate lapses {lapsed_days} days BEFORE the academic term starts!"
        }
    return {
        "status": "VERIFIED",
        "is_valid": True,
        "lapsed_days_before_term": 0,
        "alert": "Certificate remains active throughout the academic term."
    }

8. Gemini Multimodal Blueprint Parser (`services/gemini_service.py`)

Multimodal Foundation AI
What It Does: Sends architectural floor plans to Google Gemini 1.5 Flash. Extracts room labels, dimensions, and total carpet area where traditional OCR fails.
View Python Source Code 
import google.generativeai as genai
from PIL import Image
import io, json

def extract_blueprint_dimensions(image_bytes: bytes, api_key: str) -> dict:
    genai.configure(api_key=api_key)
    model = genai.GenerativeModel('gemini-1.5-flash')
    img = Image.open(io.BytesIO(image_bytes))
    
    prompt = """
    You are an AICTE Architectural Inspection Specialist.
    Inspect this architectural floor plan and extract:
    1. Labeled room names (Computer Lab, Workshop, Classrooms, Library).
    2. Dimensions and calculated carpet area in square meters.
    Return strictly JSON with this schema:
    {
      "rooms": [{"name": "Computer Lab", "area_sqm": 410}],
      "total_carpet_area_sqm": 1250
    }
    """
    response = model.generate_content([prompt, img])
    raw_text = response.text.strip()
    
    clean_json = raw_text.replace("```json", "").replace("```", "").strip()
    try:
        parsed = json.loads(clean_json)
        return {
            "rooms": parsed.get("rooms", []),
            "total_carpet_area_sqm": parsed.get("total_carpet_area_sqm", 0.0),
            "raw_response": raw_text
        }
    except Exception:
        return {"rooms": [], "total_carpet_area_sqm": 0.0, "raw_response": raw_text}
